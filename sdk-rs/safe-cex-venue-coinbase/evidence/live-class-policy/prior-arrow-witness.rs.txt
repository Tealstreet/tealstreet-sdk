
#[test]
fn prior_transplanted_supported_arrow_retains_origin() {
    let q = Microtasks::default();
    let (origin, _) = policy_class(false, json!({"key":"fixture-key","secret":"fixture-secret"}), &q);
    *origin.selected_portfolio.borrow_mut() = Some(Rc::new(RefCell::new(serde_json::from_value(json!({"uuid":"origin-cfm","name":"origin","type":"CFM","deleted":false})).unwrap())));
    origin.support_spot.set(false);
    let (target, _) = policy_class(false, json!({}), &q);
    *target.get_supported_types.borrow_mut() = origin.get_supported_types.borrow().clone();
    drop(origin.host.captured_finder.borrow_mut().take());
    drop(origin.host.captured_supported.borrow_mut().take());
    drop(target.host.captured_finder.borrow_mut().take());
    drop(target.host.captured_supported.borrow_mut().take());
    drop(origin);
    assert_eq!(&*policy_value(target.supported_types(), &q).borrow(), &["cfm"], "native transplanted supported arrow retains original receiver");
    assert!(target.selected_portfolio.borrow().is_none(), "native target selection stays untouched");
}
#[test]
fn prior_transplanted_finder_retains_origin() {
    let q = Microtasks::default();
    let (origin, leaves) = policy_class(false, json!({"key":"fixture-key","secret":"fixture-secret"}), &q);
    let expected = leaves.portfolios.borrow()[0].clone();
    expected.borrow_mut().uuid = "origin-intx".into();
    let (target, target_leaves) = policy_class(false, json!({"key":"fixture-key","secret":"fixture-secret"}), &q);
    target_leaves.portfolios.borrow()[0].borrow_mut().uuid = "target-intx".into();
    *target.find_selected_portfolio.borrow_mut() = origin.find_selected_portfolio.borrow().clone();
    drop(origin.host.captured_finder.borrow_mut().take());
    drop(origin.host.captured_supported.borrow_mut().take());
    drop(target.host.captured_finder.borrow_mut().take());
    drop(target.host.captured_supported.borrow_mut().take());
    drop(origin);
    let selected = policy_portfolio(target.find_selected(), &q);
    assert_eq!(selected.borrow().uuid, "origin-intx", "native transplanted finder reads original receiver HTTP leaves");
    assert!(Rc::ptr_eq(&selected, &expected), "native transplanted finder returns original portfolio root");
    policy_calls(&target_leaves, &[]);
}
#[test]
fn prior_transplanted_arrow_retains_live_origin_option_slot() {
    let q = Microtasks::default();
    let (origin, _) = policy_class(false, json!({"key":"fixture-key","secret":"fixture-secret"}), &q);
    *origin.selected_portfolio.borrow_mut() = Some(Rc::new(RefCell::new(serde_json::from_value(json!({"uuid":"origin-cfm","name":"origin","type":"CFM","deleted":false})).unwrap())));
    origin.support_spot.set(false);
    let weak = Rc::downgrade(&origin);
    let (target, _) = policy_class(false, json!({}), &q);
    *target.get_supported_types.borrow_mut() = origin.get_supported_types.borrow().clone();
    drop(origin.host.captured_finder.borrow_mut().take());
    drop(origin.host.captured_supported.borrow_mut().take());
    drop(target.host.captured_finder.borrow_mut().take());
    drop(target.host.captured_supported.borrow_mut().take());
    drop(origin);
    let retained = weak.upgrade().expect("native transplanted arrow retains origin after external root drop");
    *retained.options.borrow_mut() = Frame::from_json(&json!({}));
    assert_eq!(&*policy_value(target.supported_types(), &q).borrow(), &["intx"], "native transplanted arrow reads replaced origin options");
    *retained.options.borrow_mut() = Frame::from_json(&json!({"key":"fixture-key","secret":"fixture-secret"}));
    assert_eq!(&*policy_value(target.supported_types(), &q).borrow(), &["cfm"], "native transplanted arrow rereads restored origin options");
}
#[test]
fn prior_reassigned_arrow_keeps_old_code_and_origin() {
    let q = Microtasks::default();
    let (origin, _) = policy_class(false, json!({"key":"fixture-key","secret":"fixture-secret"}), &q);
    *origin.selected_portfolio.borrow_mut() = Some(Rc::new(RefCell::new(serde_json::from_value(json!({"uuid":"origin-cfm","name":"origin","type":"CFM","deleted":false})).unwrap())));
    origin.support_spot.set(false);
    let (target, _) = policy_class(false, json!({}), &q);
    let (replacement, _) = policy_class(true, json!({}), &q);
    *replacement.selected_portfolio.borrow_mut() = Some(Rc::new(RefCell::new(serde_json::from_value(json!({"uuid":"replacement-cfm","name":"replacement","type":"CFM","deleted":false})).unwrap())));
    let original = origin.capture_supported();
    *target.get_supported_types.borrow_mut() = origin.get_supported_types.borrow().clone();
    *origin.get_supported_types.borrow_mut() = replacement.get_supported_types.borrow().clone();
    assert_eq!(&*policy_value(origin.supported_types(), &q).borrow(), &["spot"]);
    *origin.get_supported_types.borrow_mut() = target.get_supported_types.borrow().clone();
    assert_eq!(&*policy_value(origin.supported_types(), &q).borrow(), &["cfm"]);
    assert_eq!(&*policy_value(target.supported_types(), &q).borrow(), &["cfm"], "native reassigned arrow retains original code and receiver");
    assert!(target.capture_supported().same_function(&original));
}
